cd /d C:\SummitTale\Servidor

del /q run_stdout.txt run_stderr.txt run_exitcode.txt 2>nul

powershell -NoProfile -Command "$p = Start-Process -FilePath '.\Server.exe' -WorkingDirectory 'C:\SummitTale\Servidor' -RedirectStandardOutput 'C:\SummitTale\Servidor\run_stdout.txt' -RedirectStandardError 'C:\SummitTale\Servidor\run_stderr.txt' -PassThru -Wait; $p.ExitCode | Out-File 'C:\SummitTale\Servidor\run_exitcode.txt' -Encoding ascii"

echo.
echo ========== EXIT CODE ==========
type run_exitcode.txt

echo.
echo ========== STDOUT ==========
type run_stdout.txt

echo.
echo ========== STDERR ==========
type run_stderr.txt

echo.
echo ========== EVENTO DE ERRO MAIS RECENTE ==========
powershell -NoProfile -Command "Get-WinEvent -FilterHashtable @{LogName='Application'; Level=2} -MaxEvents 30 | Where-Object { $_.Message -match 'Server.exe' } | Select-Object -First 1 TimeCreated,Id,ProviderName,Message | Format-List"
